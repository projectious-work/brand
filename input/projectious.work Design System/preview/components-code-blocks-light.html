12# Validate deploy policy before staging rollout
from dataclasses import dataclass
from typing import Iterable

@dataclass
class Policy:
    name: str
    max_cost_usd: float = 250.0

def enforce(policies: Iterable[Policy]) -> bool:
    return all(p.max_cost_usd <= 500 for p in policies)

12// Execute a single agent step with a 30s timeout
use tokio::time::{timeout, Duration};
use anyhow::Result;

#[derive(Debug)]
pub struct Step { pub id: u64, pub name: String }

pub async fn run(step: &Step) -> Result<String> {
    let out = timeout(Duration::from_secs(30), execute(step)).await??;
    Ok(format!("step {} ok: {}", step.id, out))
}
